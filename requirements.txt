# ============================================================================
# Paper 2 -- CAGE Challenge 2 (CybORG v2.1) pinned environment.
# CybORG is VERSION-SENSITIVE. We do NOT pip-install CybORG: it is pure-Python and
# its setup.py egg_info breaks on modern pip. Instead we clone the CAGE 2 repo and
# add it to sys.path at run time (see STAGE1_RUN_ON_COLAB.md / scripts). This file
# pins only the libraries CybORG imports.
# ----------------------------------------------------------------------------
# Why these exact versions:
#   * gym is UNPINNED upstream -> pip would grab gym>=0.26 whose 5-tuple step API
#     breaks CybORG's wrappers. We need the OLD API, so gym < 0.26.
#     gym 0.21.0 is the "challenge era" pin but FAILS to build on modern pip
#     (metadata-generation-failed). gym 0.23.1 has the same old API AND a clean
#     wheel -> use it.
#   * CybORG 2.1 uses NO removed numpy aliases (np.float/np.bool), verified by
#     grep, so it does NOT need numpy < 1.24. We use numpy 1.26.4 (has a wheel on
#     Colab's Python; gym 0.23.1 + SB3 both need numpy < 2).
# ----------------------------------------------------------------------------
gym==0.23.1
numpy==1.26.4

# CybORG's own runtime deps (installed explicitly because we use --no-deps):
paramiko
pyyaml
prettytable
docutils
